import json
import os
import sqlite3
from datetime import datetime, timezone
from pathlib import Path
from dotenv import load_dotenv

load_dotenv()
DB_PATH = os.getenv("DATABASE_PATH", "railway_history.db")

def connect():
    Path(DB_PATH).parent.mkdir(parents=True, exist_ok=True) if Path(DB_PATH).parent != Path(".") else None
    conn = sqlite3.connect(DB_PATH, timeout=30)
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    with connect() as conn:
        conn.execute("""
            CREATE TABLE IF NOT EXISTS train_history (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                collected_at TEXT NOT NULL,
                station_code TEXT,
                platform_number INTEGER,
                train_number TEXT,
                train_name TEXT,
                status TEXT,
                delay_minutes REAL,
                current_station TEXT,
                next_station TEXT,
                distance_to_next_km REAL,
                payload_json TEXT NOT NULL
            )
        """)
        conn.execute("CREATE INDEX IF NOT EXISTS idx_history_station_platform ON train_history(station_code, platform_number)")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_history_train_time ON train_history(train_number, collected_at)")
        conn.commit()

def _number(value):
    try:
        if value is None or value == "":
            return None
        return float(value)
    except (ValueError, TypeError):
        return None

def _first(d, *keys):
    if not isinstance(d, dict):
        return None
    for key in keys:
        val = d.get(key)
        if val is not None:
            return val
    return None

def save_snapshot(data, station_code=None, platform_number=None):
    """Save a raw provider payload or normalized platform response."""
    init_db()
    now = datetime.now(timezone.utc).isoformat()
    trains = data.get("trains") if isinstance(data, dict) else None
    if not isinstance(trains, list):
        # A live train payload may itself describe one train.
        trains = [data] if isinstance(data, dict) else []
    saved = 0
    with connect() as conn:
        for train in trains:
            if not isinstance(train, dict):
                continue
            train_number = _first(train, "train_number", "trainNumber", "number", "train_no")
            train_name = _first(train, "train_name", "trainName", "name")
            status = _first(train, "status", "running_status", "runningStatus")
            delay = _number(_first(train, "delay_minutes", "delayMinutes", "delay", "late_by"))
            current = _first(train, "current_station", "currentStation", "current_location", "currentLocation")
            nxt = _first(train, "next_station", "nextStation", "next_stop")
            distance = _number(_first(train, "distance_to_next_km", "distanceToNextStationKm", "distance_to_next_station_km"))
            conn.execute("""
                INSERT INTO train_history
                (collected_at, station_code, platform_number, train_number, train_name, status,
                 delay_minutes, current_station, next_station, distance_to_next_km, payload_json)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
            """, (now, station_code, platform_number, str(train_number) if train_number is not None else None,
                  str(train_name) if train_name is not None else None, str(status) if status is not None else None,
                  delay, str(current) if current is not None else None, str(nxt) if nxt is not None else None,
                  distance, json.dumps(train, ensure_ascii=False)))
            saved += 1
        conn.commit()
    return saved

def recent_history(station=None, platform=None, train_number=None, limit=100):
    init_db()
    query = "SELECT * FROM train_history WHERE 1=1"
    args = []
    if station:
        query += " AND station_code = ?"; args.append(station.upper())
    if platform is not None:
        query += " AND platform_number = ?"; args.append(int(platform))
    if train_number:
        query += " AND train_number = ?"; args.append(str(train_number))
    query += " ORDER BY id DESC LIMIT ?"; args.append(max(1, min(int(limit), 1000)))
    with connect() as conn:
        rows = conn.execute(query, args).fetchall()
    result = []
    for row in rows:
        item = dict(row)
        try: item["payload"] = json.loads(item.pop("payload_json"))
        except Exception: item["payload"] = item.pop("payload_json", None)
        result.append(item)
    return result

def provider_get(url):
    headers = {"Accept": "application/json"}
    if API_KEY:
        headers["Authorization"] = f"Bearer {API_KEY}"
        headers["x-api-key"] = API_KEY
    try:
        response = requests.get(url, headers=headers, timeout=TIMEOUT)
        response.raise_for_status()
        return response.json()
    except requests.RequestException as exc:
        raise HTTPException(status_code=502, detail=f"Railway data provider request failed: {exc}")
    except ValueError:
        raise HTTPException(status_code=502, detail="Railway data provider returned invalid JSON")

def unwrap(data):
    """Handle common API response envelopes without assuming one exact schema."""
    if not isinstance(data, dict):
        return data
    for key in ("data", "result", "response"):
        value = data.get(key)
        if isinstance(value, (dict, list)):
            return value
    return data

def train_list(data):
    data = unwrap(data)
    if isinstance(data, list):
        return data
    if isinstance(data, dict):
        for key in ("trains", "results", "items", "data"):
            if isinstance(data.get(key), list):
                return data[key]
    return []

def find_value(obj, *keys):
    if not isinstance(obj, dict):
        return None
    for key in keys:
        if obj.get(key) is not None:
            return obj[key]
    return None

@app.get("/")

@app.get("/api/stations/{station_code}/platform/{platform_number}")
def platform_trains(station_code: str, platform_number: int):
    station = station_code.upper()
    if PLATFORM_TEMPLATE:
        url = PLATFORM_TEMPLATE.format(station=station, platform=platform_number)
        data = provider_get(url)
        data = unwrap(data)
        trains = train_list(data)
        response = {"success": True, "station": station, "platform": platform_number,
                    "train_count": len(trains), "trains": trains, "source": "provider"}
        save_snapshot(response, station, platform_number)
        return response
    # Safe fallback: return what has actually been collected, not fabricated live trains.
    rows = recent_history(station=station, platform=platform_number, limit=100)
    latest_by_train = {}
    for row in rows:
        key = row.get("train_number") or f"unknown-{row['id']}"
        latest_by_train.setdefault(key, row)
    trains = []
    for row in latest_by_train.values():
        payload = row.get("payload") or {}
        if isinstance(payload, dict):
            item = dict(payload)
            item.setdefault("train_number", row.get("train_number"))
            item.setdefault("train_name", row.get("train_name"))
            item.setdefault("status", row.get("status"))
            item.setdefault("delay_minutes", row.get("delay_minutes"))
            trains.append(item)
    return {"success": True, "station": station, "platform": platform_number,
            "train_count": len(trains), "trains": trains, "source": "local_history",
            "notice": "Configure PLATFORM_API_URL_TEMPLATE for provider live platform data."}

@app.get("/api/trains/{train_number}/intelligence")
def train_intelligence(train_number: str):

    if not isinstance(raw, dict):
        raw = {"raw": raw}
    route = find_value(raw, "route", "stations", "stops") or []
    current = find_value(raw, "current_location", "currentLocation", "current_station", "currentStation", "current")
    previous = find_value(raw, "previous_station", "previousStation", "last_station", "lastStation")
    nxt = find_value(raw, "next_station", "nextStation", "next_stop", "nextStop")
    delay = find_value(raw, "delay_minutes", "delayMinutes", "delay", "late_by")
    return {
        "success": True,
        "train_number": str(find_value(raw, "train_number", "trainNumber", "number") or train_number),
        "train_name": find_value(raw, "train_name", "trainName", "name"),
        "status": find_value(raw, "status", "running_status", "runningStatus"),
        "is_live": True,
        "from": find_value(raw, "from", "source", "origin"),
        "to": find_value(raw, "to", "destination", "dest"),
        "current_location": current,
        "previous_station": previous,
        "next_station": nxt,
        "delay_minutes": delay,
        "route_stop_count": len(route) if isinstance(route, list) else None,
        "last_updated": find_value(raw, "last_updated", "lastUpdated", "updated_at", "updatedAt"),

    }

@app.get("/api/trains/{train_number}/prediction")
def predict_delay(train_number: str):
    if not Path(MODEL_PATH).exists():
        raise HTTPException(status_code=503, detail="Model not found. Collect historical data and run: python train_model.py")
    try:
        bundle = joblib.load(MODEL_PATH)
        model = bundle["model"]
        features = bundle["features"]
    except Exception as exc:
        raise HTTPException(status_code=500, detail=f"Could not load model bundle: {exc}")
    rows = recent_history(train_number=train_number, limit=1)
    if not rows:
        raise HTTPException(status_code=404, detail="No historical snapshot exists for this train yet.")
    row = rows[0]
    # The baseline feature set is intentionally explicit and robust to missing fields.
    feature_values = {
        "platform_number": row.get("platform_number"),
        "delay_minutes": row.get("delay_minutes"),
        "distance_to_next_km": row.get("distance_to_next_km"),
        "station_code_NDLS": 1 if row.get("station_code") == "NDLS" else 0,
        "station_code_ANVT": 1 if row.get("station_code") == "ANVT" else 0,
    }
    import pandas as pd
    x = pd.DataFrame([{f: feature_values.get(f, 0) for f in features}])
    prediction = float(model.predict(x)[0])
    return {"success": True, "train_number": train_number,
            "predicted_delay_minutes": round(prediction, 2),
            "model": "trained scikit-learn model",
            "note": "Experimental estimate from historical observations; not an official ETA."}

import os
import time
import requests
from dotenv import load_dotenv
from database import save_snapshot

load_dotenv()
TEMPLATE = os.getenv("PLATFORM_API_URL_TEMPLATE", "").strip()
INTERVAL = int(os.getenv("COLLECTION_INTERVAL_SECONDS", "7200"))
BASE_URL = os.getenv("RAILRADAR_BASE_URL", "https://api.railradar.in/v1").rstrip("/")
API_KEY = os.getenv("RAILRADAR_API_KEY", "").strip()
STATIONS = os.getenv("DEFAULT_STATIONS", "NDLS:4,ANVT:1")

def fetch(url):
    headers = {"Accept": "application/json"}
    if API_KEY:
        headers["Authorization"] = f"Bearer {API_KEY}"
        headers["x-api-key"] = API_KEY
    response = requests.get(url, headers=headers, timeout=20)
    response.raise_for_status()
    return response.json()

def unwrap(data):
    if isinstance(data, dict):
        for k in ("data", "result", "response"):
            if isinstance(data.get(k), (dict, list)):
                return data[k]
    return data

def get_trains(data):
    data = unwrap(data)
    if isinstance(data, list):
        return data
    if isinstance(data, dict):
        for k in ("trains", "results", "items"):
            if isinstance(data.get(k), list):
                return data[k]
    return []

def main():
    if not TEMPLATE:
        raise SystemExit("Set PLATFORM_API_URL_TEMPLATE in .env to your valid provider platform endpoint.")
    pairs = []
    for token in STATIONS.split(","):
        if ":" in token:
            station, platform = token.strip().split(":", 1)
            pairs.append((station.upper(), int(platform)))
    print(f"Collector running for {pairs}; interval={INTERVAL}s. Ctrl+C to stop.")
    while True:
        for station, platform in pairs:
            try:
                url = TEMPLATE.format(station=station, platform=platform)
                trains = get_trains(fetch(url))
                saved = save_snapshot({"trains": trains}, station, platform)
                print(f"{station} platform {platform}: {len(trains)} trains, saved {saved}")
            except Exception as exc:
                print(f"Collection failed for {station}/{platform}: {exc}")
        time.sleep(max(60, INTERVAL))

if __name__ == "__main__":
    main()

import json
import os
import sqlite3
from pathlib import Path
import joblib
import pandas as pd
from dotenv import load_dotenv
from sklearn.ensemble import ExtraTreesRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score

load_dotenv()
DB_PATH = os.getenv("DATABASE_PATH", "railway_history.db")
MODEL_PATH = os.getenv("MODEL_PATH", "railway_delay_model.pkl")
META_PATH = os.getenv("MODEL_METADATA_PATH", "model_metadata.json")
FEATURES = ["platform_number", "distance_to_next_km", "station_code_NDLS", "station_code_ANVT"]

def main():
    if not Path(DB_PATH).exists():
        raise SystemExit(f"Database not found: {DB_PATH}. Run the backend/collector first.")
    with sqlite3.connect(DB_PATH) as conn:
        df = pd.read_sql_query("""
            SELECT collected_at, station_code, platform_number, delay_minutes, distance_to_next_km
            FROM train_history
            WHERE delay_minutes IS NOT NULL
            ORDER BY collected_at ASC
        """, conn)
    if len(df) < 20:
        raise SystemExit(f"Only {len(df)} labeled rows found. Collect at least 20; preferably hundreds across different trains and times.")
    df["station_code_NDLS"] = (df["station_code"] == "NDLS").astype(int)
    df["station_code_ANVT"] = (df["station_code"] == "ANVT").astype(int)
    for col in FEATURES:
        df[col] = pd.to_numeric(df[col], errors="coerce")
    df[FEATURES] = df[FEATURES].fillna(0)
    X, y = df[FEATURES], pd.to_numeric(df["delay_minutes"], errors="coerce")
    valid = y.notna()
    X, y = X.loc[valid], y.loc[valid]
    split = max(1, int(len(X) * 0.8))
    if len(X) - split < 1:
        split = len(X) - 1
    X_train, X_test = X.iloc[:split], X.iloc[split:]
    y_train, y_test = y.iloc[:split], y.iloc[split:]
    model = ExtraTreesRegressor(n_estimators=250, random_state=42, min_samples_leaf=2)
    model.fit(X_train, y_train)
    predictions = model.predict(X_test)
    metrics = {
        "rows": int(len(df)),
        "train_rows": int(len(X_train)),
        "test_rows": int(len(X_test)),
        "mae_minutes": round(float(mean_absolute_error(y_test, predictions)), 3),
        "rmse_minutes": round(float(mean_squared_error(y_test, predictions) ** 0.5), 3),
        "r2": round(float(r2_score(y_test, predictions)), 3) if len(y_test) >= 2 else None,
        "features": FEATURES,
        "warning": "Chronological holdout is only a first check. Sparse or biased historical data can make metrics unstable."
    }
    joblib.dump({"model": model, "features": FEATURES}, MODEL_PATH)
    Path(META_PATH).write_text(json.dumps(metrics, indent=2))
    print(json.dumps(metrics, indent=2))
    print(f"Saved model: {MODEL_PATH}")

if __name__ == "__main__":
    main()

        <div className="form-row train-search"><label>Train number<input value={trainNumber} onChange={e => setTrainNumber(e.target.value)} placeholder="12616"/></label><button onClick={loadTrain}>Analyze train</button></div>
        {intelligence && <div className="intel">
          <h4>{intelligence.train_name || `Train ${intelligence.train_number}`}</h4>
          <p className="muted">#{intelligence.train_number} · {intelligence.status || "Status unavailable"}</p>
          <div className="detail-grid">
            <div><span>Current location</span><strong>{typeof intelligence.current_location === "object" ? JSON.stringify(intelligence.current_location) : intelligence.current_location || "—"}</strong></div>
            <div><span>Previous station</span><strong>{typeof intelligence.previous_station === "object" ? JSON.stringify(intelligence.previous_station) : intelligence.previous_station || "—"}</strong></div>
            <div><span>Next station</span><strong>{typeof intelligence.next_station === "object" ? JSON.stringify(intelligence.next_station) : intelligence.next_station || "—"}</strong></div>
            <div><span>Current delay</span><strong>{intelligence.delay_minutes ?? "—"} min</strong></div>
          </div>
          {prediction && <div className="prediction"><span>Experimental predicted delay</span><strong>{prediction.predicted_delay_minutes !== undefined ? `${prediction.predicted_delay_minutes} min` : "Unavailable"}</strong><small>{prediction.note || prediction.error}</small></div>}
        </div>}
        {!intelligence && <p className="muted helper">Enter a train number to load live intelligence and an available model prediction.</p>}
      </div>
      <div className="panel">
        <p className="eyebrow">HISTORICAL STORAGE</p><h3>Recent observations</h3>
        <p className="muted">Latest stored snapshots for {station || "selected station"} · Platform {platform || "—"}</p>
        <div className="history-list">
          {history.slice(0, 8).map(row => <div className="history-row" key={row.id}><div><strong>{row.train_name || `Train ${row.train_number || "—"}`}</strong><small>{row.train_number || "No number"} · {row.collected_at ? new Date(row.collected_at).toLocaleString() : "Time unavailable"}</small></div><span>{row.delay_minutes == null ? "—" : `${row.delay_minutes} min`}</span></div>)}
          {!history.length && <div className="empty compact">No stored observations yet.</div>}
        </div>
      </div>
    </section>
    <footer>Railway Intelligence · Prototype dashboard · Predictions are experimental and not official railway information.</footer>
  </main>;
}
